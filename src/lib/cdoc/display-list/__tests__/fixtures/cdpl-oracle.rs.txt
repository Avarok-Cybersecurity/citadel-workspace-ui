// The oracle that wrote goldens.json and hundred-pages.bin. Not compiled here: build it as a
// scratch crate beside a citadel-docs checkout (own CARGO_TARGET_DIR), with
//   [dependencies] cdoc-paint = { path = "<docs>/crates/cdoc-paint" }
//   cdoc-layout = { path = "<docs>/crates/cdoc-layout", features = ["test-support"] }
//   cdoc-model = { path = "<docs>/crates/cdoc-model" }
// then `cdpl-oracle emit hundred-pages.bin` and `cdpl-oracle report <file.bin>...`.
//! Scratch oracle: writes a 100-page display list and prints, for each list given, per-page
//! op counts and checksums computed by cdoc-paint's own decoder.
use cdoc_layout::{layout, LayoutOptions};
use cdoc_model::{Block, Document, Inline, Paragraph, RunProperties};
use cdoc_paint::{decode, encode, OpView};
use std::fmt::Write as _;

const WORDS: [&str; 16] = ["the","display","list","is","one","description","of","every","page","painted","by","canvas","preview","and","file","alike"];

fn document(paragraphs: usize) -> Document {
    let mut doc = Document::blank();
    let template = match &doc.sections[0].blocks[0] { Block::Paragraph(p) => p.properties.clone(), Block::Table(_) => unreachable!() };
    let mut state: u64 = 0x2545_F491_4F6C_DD1D;
    let mut word = || { state = state.wrapping_mul(6_364_136_223_846_793_005).wrapping_add(1); WORDS[usize::try_from(state >> 60).unwrap_or(0)] };
    doc.sections[0].blocks = (0..paragraphs).map(|_| {
        let text: Vec<&str> = (0..60).map(|_| word()).collect();
        Block::Paragraph(Paragraph { properties: template.clone(), inlines: vec![Inline::text(text.join(" "), RunProperties::none())] })
    }).collect();
    doc
}

fn report(name: &str, bytes: &[u8]) -> String {
    let view = decode(bytes).expect("decodes");
    let mut out = format!("{{\"name\":\"{name}\",\"faces\":{},\"pages\":[", view.faces().len());
    for slot in 0..view.page_count() {
        let page = view.page(slot).expect("page");
        let mut c = [0u64; 11];
        let (mut text_sum, mut glyphs, mut first_pens) = (0i64, 0u64, String::new());
        for op in page.ops() {
            let k = match op {
                OpView::FillRect { .. } => 0, OpView::StrokeRect { .. } => 1, OpView::Line { .. } => 2,
                OpView::FillPath { .. } => 3, OpView::StrokePath { .. } => 4,
                OpView::TextRun { head, glyphs: g } => {
                    let mut pen = head.x;
                    let record = first_pens.is_empty();
                    for gl in g { if record { let _ = write!(first_pens, "{pen},"); } text_sum += i64::from(gl.id) + i64::from(gl.advance); pen += gl.advance; glyphs += 1; }
                    if record { let _ = write!(first_pens, "{pen}"); }
                    5 }
                OpView::GlyphRun { glyphs: g, .. } => { for gl in g { text_sum += i64::from(gl.id) + i64::from(gl.advance) + i64::from(gl.x_offset) + i64::from(gl.y_offset); glyphs += 1; } 6 }
                OpView::Image { .. } => 7, OpView::Link { .. } => 8, OpView::ClipPush(_) => 9, OpView::ClipPop => 10,
            };
            c[k] += 1;
        }
        if slot > 0 { out.push(','); }
        let _ = write!(out, "{{\"number\":{},\"width\":{},\"height\":{},\"counts\":{:?},\"glyphs\":{glyphs},\"textSum\":{text_sum},\"firstRunPens\":[{first_pens}]}}", page.number, page.width, page.height, c);
    }
    out.push_str("]}");
    out
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    if args[1] == "emit" {
        let fonts = cdoc_layout::test_support::default_fonts();
        let mut n = 400;
        let laid = loop { let l = layout(&document(n), &fonts, &LayoutOptions::word_like()).unwrap(); if l.pages.len() >= 100 { break l; } n += 40; };
        let bytes = encode(&laid, &fonts).unwrap();
        std::fs::write(&args[2], &bytes).unwrap();
        eprintln!("{n} paragraphs, {} pages, {} bytes", laid.pages.len(), bytes.len());
    } else {
        for path in &args[2..] { println!("{}", report(path.rsplit('/').next().unwrap(), &std::fs::read(path).unwrap())); }
    }
}
